const BASE_URL = "https://your-domain.com/api";

    method: "POST",
    headers: {
        "X-API-Key": "your_api_key",
        "Content-Type": "application/json"
    },
    body: JSON.stringify({
        sku: "PROD-12345",
        warehouse: "US-WEST"
    })
});

console.log(result.data);

import requests

BASE_URL = "https://your-domain.com/api"

response = requests.get(
    f"{BASE_URL}/export/inventory",
    headers={"X-API-Key": "your_api_key"},
    params={"format": "csv", "include_variants": "true"}
)

with open("inventory_export.csv", "wb") as f:
    f.write(response.content)
print("Exported inventory.csv")
const BASE_URL = "https://your-domain.com/api";

const response = await fetch(
    \`\${BASE_URL}/export/inventory?format=csv&include_variants=true\`,
    {
        headers: { "X-API-Key": "your_api_key" }
    }
);
const blob = await response.blob();
const url = window.URL.createObjectURL(blob);
const a = document.createElement("a");
a.href = url;
a.download = "inventory_export.csv";
a.click();
curl -X GET "https://your-domain.com/api/export/inventory?format=csv" \
  -H "X-API-Key: your_api_key" \
  -o inventory_export.csv
